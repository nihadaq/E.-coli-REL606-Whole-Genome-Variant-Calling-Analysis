        "import cyvcf2\n",
        "import pandas as pd\n",
        "\n",
        "vcf = cyvcf2.VCF(\"ecoli_freebayes_variants.vcf\")\n",
        "records = []\n",
        "\n",
        "for variant in vcf:\n",
        "    # Filter out low-confidence calls (QUAL < 20)\n",
        "    if variant.QUAL is None or variant.QUAL < 20:\n",
        "        continue\n",
        "\n",
        "    # Extract depth (DP)\n",
        "    dp = variant.format('DP')[0][0] if 'DP' in variant.FORMAT else variant.INFO.get('DP', None)\n",
        "\n",
        "    # Classify variant type\n",
        "    if variant.is_snp:\n",
        "        var_type = \"SNP\"\n",
        "    elif variant.is_indel:\n",
        "        var_type = \"INDEL\"\n",
        "    else:\n",
        "        var_type = str(variant.INFO.get(\"TYPE\", \"OTHER\"))\n",
        "\n",
        "    records.append({\n",
        "        \"CHROM\": variant.CHROM,\n",
        "        \"POS\": variant.POS,\n",
        "        \"REF\": variant.REF,\n",
        "        \"ALT\": \",\".join(variant.ALT),\n",
        "        \"QUAL\": round(variant.QUAL, 2),\n",
        "        \"DP\": dp,\n",
        "        \"TYPE\": var_type\n",
        "    })\n",
        "\n",
        "df = pd.DataFrame(records)\n",
        "print(f\"Total High-Quality Variants Processed: {len(df)}\")"
